fallingStar:
    push bp
    mov bp, sp
    sub sp, 2

    push ax
    push bx
    push cx
    push dx
    push di
    push es

    mov ax, [bp+4]
    dec ax
    mov bx, ax

    mov word [bp-2], 0

    mov ax, 0xb800
    mov es, ax

starloop:
    mov ax, [bp-2]
    mov dx, 80
    mul dx

    add ax, bx
    shl ax, 1
    mov di, ax

    mov word [es:di], 0x072A

    mov cx, 0xffff

delay:
    loop delay

    mov word [es:di], 0x0720

    inc word [bp-2]

    cmp word [bp-2], 25
    jne starloop

    mov word [bp-2], 0
    jmp starloop
